cmake_minimum_required(VERSION 3.25)
project(StrokeSdkProbe LANGUAGES CXX RC)
if(NOT MSVC)
    message(FATAL_ERROR "This diagnostic verifies the MSVC / Windows SDK toolchain")
endif()
add_executable(stroke_sdk_probe main.cpp probe.rc)
target_compile_features(stroke_sdk_probe PRIVATE cxx_std_20)
target_compile_options(stroke_sdk_probe PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/WX;/utf-8;/permissive->")
target_compile_definitions(stroke_sdk_probe PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_link_libraries(stroke_sdk_probe PRIVATE ole32 uuid)
